icd_culture_assimilation_mode = {
	
	
	map_color = {
		if = {
			limit = {
				has_owner = no
				is_land = yes

			}
			value = rgb { 0 0 0 }
		}
		else_if = {
			limit = {
				is_land = yes
				is_ownable = yes
				has_owner = yes
				OR = {
					province ?= {
						has_variable = promoting_culture
						var:promoting_culture = scope:actor.culture
					}
					province ?= {
						NOT = { has_variable = promoting_culture }
						owner.culture ?= scope:actor.culture
					}
				}
			}
			value = define:NMapColors|MAP_COLOR_MAX
		}
		else_if = {
			limit = {
				is_land = yes

				dominant_culture = scope:actor.culture
				OR = {
					province ?= {
						has_variable = promoting_culture
						NOT = { var:promoting_culture = scope:actor.culture }
					}
					province ?= {
						NOT = { has_variable = promoting_culture }
						NOT = { owner.culture ?= scope:actor.culture }
					}
				}
			}
			value = define:NMapColors|MAP_COLOR_LOW
		}
		else_if = {
			limit = {
				is_land = yes
				is_ownable = yes
				OR = {
					province ?= {
						has_variable = promoting_culture
						var:promoting_culture = root.owner.culture
					}
					province ?= {
						NOT = { has_variable = promoting_culture }
					}
				}
			}
			value = define:NMapColors|MAP_COLOR_HIGH
		}
		else_if = {
			limit = {
				is_land = yes
				is_ownable = yes
				province ?= {
					has_variable = promoting_culture
					var:promoting_culture = root.dominant_culture
				}
			}
			value = define:NMapColors|MAP_COLOR_MID
		}
		else_if = {
			limit = {
				is_land = yes
				is_ownable = yes

					has_variable = promoting_culture
				}
			}

icd_tolerance_heretic_map = {

	map_color = {
		if = {
			limit = {
				has_owner = yes
			}

			if = {
				limit = { 
					owner = {
						modifier:tolerance_heretic >= 1.5
					}
				}
				value = define:NMapColors|MAP_COLOR_MAX
			}
			else_if = {
				limit = { 
					owner = {
						modifier:tolerance_heretic >= 0.5
						modifier:tolerance_heretic < 1.5
					}
				}
				lerp = {
					min_color = define:NMapColors|MAP_COLOR_HIGH
					max_color = define:NMapColors|MAP_COLOR_MAX
					factor = {
						value = owner.modifier:tolerance_heretic
						subtract = 0.5
					}
				}
			}
			else_if = {
				limit = { 
					owner = {
						modifier:tolerance_heretic >= -0.5
						modifier:tolerance_heretic < 0.5
					}
				}
				lerp = {
					min_color = define:NMapColors|MAP_COLOR_MID
					max_color = define:NMapColors|MAP_COLOR_HIGH
					factor = {
						value = owner.modifier:tolerance_heretic
						add = 0.5
					}
				}
			}

				limit = { 
					owner = {

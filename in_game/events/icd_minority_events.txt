icd_minority_events.1000 = { ###expel minority cabinet action###
	hide_portraits = yes
	type = country_event
	title = icd_minority_events.1.title
	desc = icd_minority_events.1.desc
	outcome = neutral

	hidden = yes
	
	illustration_tags = {
        10 = angry
        10 = exterior
    }
	immediate = {
		event_illustration_poptype_effect = { foreground = pop_type:soldiers background = pop_type:soldiers }
		
		var:icd_expel_culture_people = {
			save_scope_as = icd_expel_scope
		}
		
		ordered_country_in_diplomatic_range = { ##check if has primary country###
			limit = {
				culture = scope:icd_expel_scope
				NOT = { 
					this = root
					any_overlord_or_above = { 
						this = root
					}
				}
			}
			order_by = total_population
			check_range_bounds = no
			max = 1
			save_scope_as = target_recipient_country
		}

		if = { ##check if has accepted country###
			limit = {
				NOT = {
					exists = scope:target_recipient_country
				}
			}
			ordered_country_in_diplomatic_range = {
				limit = {
					num_provinces >= 3
					has_primary_or_accepted_culture = scope:icd_expel_scope
					NOT = { 
						this = root
						any_overlord_or_above = { 
							this = root
						}
					}
				}
				order_by = total_population
				check_range_bounds = no
				max = 1
				save_scope_as = target_recipient_country
			}
		}
		
		if = { ##check if has kindred opinion or shared culture group###
			limit = {
				NOT = {
					exists = scope:target_recipient_country
				}
			}
			ordered_country_in_diplomatic_range = {
				limit = {
					num_provinces >= 3
					culture = {
						cultural_view = {
							 target = scope:icd_expel_scope
							 value = kindred
						}
						has_shared_culture_group = scope:icd_expel_scope
					}
					NOT = { 
						this = root
						any_overlord_or_above = { 
							this = root
						}
					}
				}
				order_by = total_population
				check_range_bounds = no
				max = 1
				save_scope_as = target_recipient_country
			}
		}
		
		if = {##jewish check###
			limit = {
				NOT = {
					exists = scope:target_recipient_country
				}
				scope:icd_expel_scope = {
					has_culture_group = culture_group:jewish_group
				}
			}
			ordered_country_in_diplomatic_range = {
				limit = {
					num_provinces >= 3
					NOT = { 
						this = root
						any_overlord_or_above = { 
							this = root
						}
					}
					OR = {
						has_estate_privilege = estate_privilege:jewish_charter
						AND = {
								has_estate_privilege = estate_privilege:icd_dhimmi_communities_protection
								NOT = { has_variable = icd_israelite_dhimmi_disabled }
							}
					}
				}
				order_by = total_population
				check_range_bounds = no
				max = 1
				save_scope_as = target_recipient_country
			}
		}
		
	}

	option = { #####
		name = icd_minority_events.1.a
		
		if = {#####
			limit = {
				exists = scope:target_recipient_country
				scope:target_recipient_country = {
					NOT = {
						has_variable = icd_expel_origin_country
					}
				}
			}
			scope:target_recipient_country = {
				trigger_event_silently = {
					id = icd_minority_events.1001
					days = 1
				}
			}
		}
		else = {#####
			trigger_event_silently = {
				id = icd_minority_events.1002
				days = 1
			}
		}
		
		scope:target_recipient_country = {
			set_variable = {
				name = icd_expel_origin_country
				value = root
			}
		}
		
	}
}

icd_minority_events.1001 = { ###checking with country###
	hide_portraits = yes
	type = country_event
	title = icd_minority_events.1001.title
	desc = icd_minority_events.1001.desc
	outcome = neutral

	
	illustration_tags = {
        10 = angry
        10 = exterior
    }
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:nobles_estate }
	}

	option = { ###accept##
		name = icd_minority_events.1001.a
		
		ai_will_select  = { value = 2000 }
		
		var:icd_expel_origin_country = {
			trigger_event_silently = {
				id = icd_minority_events.1003
				days = 1
			}
			set_variable = {
				name = icd_expel_destination_country
				value = root
			}
		}
		
		remove_variable = icd_expel_origin_country
	}
	
	option = { ###refuse##
		name = icd_minority_events.1001.b
		
		ai_will_select  = { value = -500 }
		
		var:icd_expel_origin_country = {
			trigger_event_silently = {
				id = icd_minority_events.1002
				days = 1
			}
		}
		
		remove_variable = icd_expel_origin_country
	}
}

icd_minority_events.1003 = { ###begin expulsion###
	hide_portraits = yes
	type = country_event
	title = icd_minority_events.1003.title
	desc = icd_minority_events.1003.desc
	outcome = neutral

	
	illustration_tags = {
        10 = angry
        10 = exterior
    }
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:nobles_estate }
		
		var:icd_expel_destination_country = {
			save_scope_as = target_recipient_country
		}
		
		scope:target_recipient_country = {
			capital.province = {
				save_scope_as = target_destination
			}
			random_province = {
				limit = { NOT = { this = scope:target_destination }
					NOT = { province_definition = { any_location_in_province_definition = { owner ?= root } }}
				}
				save_scope_as = target_destination_two
			}
			random_province = {
				limit = { NOT = { this = scope:target_destination }
					NOT = { this = scope:target_destination_two }
					NOT = { province_definition = { any_location_in_province_definition = { owner ?= root } }}
				}
				save_scope_as = target_destination_three
			}
		}
		
		
		
		
	}

	option = { #####
		name = icd_minority_events.1003.a
		
		var:icd_expel_culture_people = {
			save_scope_as = icd_migration_culture_scope
		}
		
		add_stability = stability_mild_penalty
		add_prestige = prestige_weak_penalty
		change_gold_effect = { scale = -25 }
		
		var:icd_expel_culture_people = {
			change_cultural_view = {
				target = root.culture
				change = -3
			}
		}
		
		custom_tooltip = icd_begin_expelling_tt
		hidden_effect = {
			every_province = {
				add_migration = {
					from = this.province_definition
					to = scope:target_destination.province_definition
					culture = scope:icd_migration_culture_scope
					amount = {
						value = 0.500
					}
					months = 24
				}
			}
			every_province = {
				add_migration = {
					from = this.province_definition
					to = scope:target_destination_two.province_definition
					culture = scope:icd_migration_culture_scope
					amount = {
					value = 0.500
					}
					months = 24
				}
			}
			every_province = {
				add_migration = {
					from = this.province_definition
					to = scope:target_destination_three.province_definition
					culture = scope:icd_migration_culture_scope
					amount = {
					value = 0.500
					}
					months = 24
				}
			}
		}
		
		remove_variable = icd_expel_culture_people
		remove_variable = icd_expel_destination_country
	}
	
	option = { ##on second thought###
		name = icd_minority_events.1003.b
		
		remove_variable = icd_expel_culture_people
		remove_variable = icd_expel_destination_country
	}
}

icd_muslim_events.4 = {
	type = country_event
	title = icd_muslim_events.4.title
	desc = icd_muslim_events.4.desc
	outcome = neutral
	trigger = {
	}

	illustration_tags = {
        10 = armed
		10 = exterior
    }

	immediate = {
		event_illustration_peasants_foreign_religion_pop_type_effect = yes
		
		global_var:icd_current_caliph = {
			save_scope_as = icd_caliph_ruler
		}
		
		ordered_country_in_diplomatic_range = {
			limit = {
				num_provinces >= 3
				NOT = { 
					this = root
					any_overlord_or_above = { 
						this = root
					}
				}
				religion = religion:sunni
			}
			order_by = {
				value = "border_distance_to(root)"
				multiply = -1
			}
			max = 1
			save_scope_as = target_recipient_country
		}
		
		scope:target_recipient_country = {
			random_owned_location = {
				save_scope_as = target_location_one
			}
			random_owned_location = {
				limit = { NOT = { this = scope:target_location_one } }
				save_scope_as = target_location_two
			}
			random_owned_location = {
				limit = { NOT = { this = scope:target_location_one } 
					NOT = { this = scope:target_location_two } 
				}
				save_scope_as = target_location_three
			}
		}
	}

	option = {
		name = icd_muslim_events.4.a
		
		custom_tooltip = {
			text = icd_caliphate_fatwa_hijra_tt
			every_owned_location = {
				limit = {
					"religion_population(religion:sunni)" > 0.5
				}
				add_location_modifier = {
					modifier = icd_caliphate_fatwa_hijra
					years = 10
					mode = add_and_extend
				}
				
				add_migration = {
					from = this.province_definition
					to = scope:target_location_one.province_definition
					religion = religion:sunni
					amount = {
						value = 0.015
					}
					months = 12
				}
				add_migration = {
					from = this.province_definition
					to = scope:target_location_two.province_definition
					religion = religion:sunni
					amount = {
						value = 0.015
					}
					months = 12
				}
				add_migration = {
					from = this.province_definition
					to = scope:target_location_three.province_definition
					religion = religion:sunni
					amount = {
						value = 0.015
					}
					months = 12
				}
			}
		}

	type = country_event
	title = icd_muslim_events.27.title
	desc = icd_muslim_events.27.desc
	outcome = neutral

	trigger = {
		religion = religion:zaydi_religion
		NOT = { has_variable = icd_zaydi_reform_four }	
	}

	illustration_tags = {
        10 = armed
		10 = exterior
    }

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
		
		create_rebel = {
			category = pretender
			name = imamate_pretenders
			save_scope_as = target_rebel
		}

	}

	option = { 
		name = icd_muslim_events.27.a
		
		custom_tooltip = khuruj_allegiance_changes
		
		hidden_effect = {
			random_character  = {
				limit = { estate_type = estate_type:nobles_estate }
				save_scope_as = target_character
				change_character_allegiance = scope:target_rebel
			}
			ordered_province = {
				limit = { dominant_religion = religion:zaydi_religion }
				order_by = {
					value = province_satisfaction
					multiply = -1
				}
				max = 1
				every_location_in_province = {
					every_pop = {
						change_pop_allegiance = scope:target_rebel
					}
				}
			}
		}
		
		add_country_modifier = {
			modifier = icd_khuruj_modifier
			mode = extend
			years = {
				value = 15
				if = {
					limit = { has_variable = icd_zaydi_reform_one }
					subtract = 5
				}
				if = {
					limit = { has_variable = icd_zaydi_reform_two }
					subtract = 5
				}
				if = {
					limit = { has_variable = icd_zaydi_reform_three }
					subtract = 5
				}
			}
		}
	}
	
}

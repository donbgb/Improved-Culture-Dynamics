namespace = icd_radical_reformation

scripted_effect add_to_reformer_list = {
	add_to_global_variable_list = {
		name = reformer_list
		target = scope:target_character
	}
}

icd_radical_reformation.1 = { #Zwickau prophets
	type = country_event
	category = situation_event
	title = icd_radical_reformation.1.title
	desc = icd_radical_reformation.1.desc
	outcome = neutral

	fire_only_once = yes

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:clergy_estate }

		character:martin_luther = {
			save_scope_as = target_character_luther
		}
		
		character:reformer_philip_melanchton = {
			save_scope_as = target_character_melanchton
		}
		
		religion:anabaptist = {
			enable_religion = yes
		}
	}
	
	option = {
		name = icd_radical_reformation.1.a
		
		location:wittenberg = {
			every_pop = {
				limit = {
					OR = {
						religion = religion:lutheran
						religion = religion:catholic
						religion = religion:calvinist
					}
				}
				split_pop = {
					fraction = 0.05
					religion = religion:anabaptist
				}
			}
		}
	}
}
	
icd_radical_reformation.2 = { #Swiss Brethren and Conrad Grebel#
	type = country_event
	category = situation_event
	title = icd_radical_reformation.2.title
	desc = icd_radical_reformation.2.desc
	outcome = neutral

	fire_only_once = yes

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:clergy_estate }

		create_character = {
			first_name = name_conrad # 
			last_name = Grebel
			religion = religion:anabaptist
			adm = { 60 80 }
			dip = { 60 80 }
			mil = { 60 80 }
			birth_date = 1498.1.1
			birth_location = location:zurich
			culture = culture:high_alemannic
			estate = estate_type:burghers_estate
			save_scope_as = target_character
			script = swi_conrad_grebel
			add_trait = trait:unsuited_for_naval_command
			add_trait = trait:unsuited_for_army_command
		}
		
		
		
		add_to_reformer_list = yes
		situation:reformation = {
			var:lutheran_character = {
				save_scope_as = target_luther
			}
		}
		location:zurich = {
			save_scope_as = target_location
			set_variable = received_anabaptist
		}
		random_neighbor_country = {
			save_scope_as = exile_destination
		}
		
		scope:target_location = {
			spawn_movement = {
				movement_definition = movement_definition:anabaptist_movement
				supporters = {
					value = population
					multiply = 0.01
				}
			}
		}

		religion:anabaptist = {
			ordered_movement_in_religion = {
				max = 1
				order_by = { value = 1 }
				add_spreader = {
					character = scope:target_character
					location = scope:target_location
				}
			}
		}
		
		trigger_event_silently = {
			id = icd_radical_reformation.3
			years = 1
		}
		
		character:swi_huldrych_zwingli = {
			save_scope_as = icd_reformer_rude
		}
	}
	
	option = {
		name = icd_radical_reformation.2.a ##okay##
		
		ai_will_select = {
			value = -500
		}
	}
	
	option = {
		name = icd_radical_reformation.2.b ##exile##

		scope:target_character = {
			move_country = scope:exile_destination
		}
		
		location:zurich = {
			add_location_modifier = {
				modifier = icd_expelled_anabaptists
				years = 20
			}
		}
	}
	
	option = {
		name = icd_radical_reformation.2.c ##kill##

		scope:target_character = {
			kill_character = scope:target_character
		}
		
		location:zurich = {
			add_location_modifier = {
				modifier = icd_expelled_anabaptists
				years = 20
			}
		}
	}
}

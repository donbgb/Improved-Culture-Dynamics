icd_reformation_push.2 = {
	type = country_event
	title = icd_reformation_push.2.title
	desc = icd_reformation_push.2.desc
	outcome = neutral


	fire_only_once = yes
	dynamic_historical_event = {
		tag = TEU
		from = 1526.1.1
		to = 1570.1.1
		monthly_chance = 10
	}

	trigger = {
		is_subject = no
		at_war = no
		government_type = government_type:theocracy
		num_locations > 1
		
		capital = {
			OR = {
				region = region:baltic_region
				area = area:pomerania_area
				area = area:brandenburg_area
				area = area:silesia_area
			}
		}
		
		NOT = {
			any_neighbor_country = {
				religion.group = religion_group:folk_european_group
			}
			c:LIT = {
				religion.group = religion_group:folk_european_group
			}
			c:POL = {
				religion.group = religion_group:folk_european_group
			}
			country_exists = c:PRU
		}
		religion = religion:catholic
	}

	illustration_tags = {
		10 = angry
		10 = interior
	}

	immediate = {
		event_illustration_poptype_effect = { foreground = pop_type:clergy background = pop_type:soldiers }
	
	}

	option = {
		name = icd_reformation_push.2.a  ##convert##
		
		if = {
			limit = {
				c:LIV = { 
					government_type = government_type:theocracy
					religion = religion:catholic
				}
			}
			custom_tooltip = {
				text = icd_livonians_will_decide
				c:LIV = {
					trigger_event_silently = {
						id = icd_reformation_push.3
						days = 5
					}
				}
			}
		}
		
		revoke_estate_privilege = estate_privilege:clergy_military_orders
		remove_reform = government_reform:military_order_reform
		
		form_country = formable_country:PRU_f
		change_government_type = government_type:monarchy
		
		hidden_effect = {
			ordered_owned_location = {
				limit = { dominant_religion = religion:catholic }
				order_by = population
				max = {
					value = 2
					add = {
						value = root.num_provinces
						multiply = 0.9
					}
				}
				check_range_bounds = no
			
				every_pop = {
					limit = { religion = religion:catholic 
						culture = culture:prussian
					}
					split_pop = {
						fraction = 0.05
						religion = religion:lutheran
					}
				}
				every_pop = {
					limit = { religion = religion:catholic 
						culture = culture:prussian
						OR = {
							pop_type = pop_type:nobles
							pop_type = pop_type:clergy
						}
					}
					split_pop = {
						fraction = 0.3
						religion = religion:lutheran
					}
				}
			}
		}
		
		change_religion = religion:lutheran
		change_religion_for_ruler_and_family = { country = ROOT religion = religion:lutheran }
		every_character = {
			limit = {
				religion = religion:catholic 
				culture = culture:prussian
				OR = {
					has_estate = estate_type:clergy_estate
					has_estate = estate_type:nobles_estate
				}
			}
			change_character_religion = religion:lutheran
		}
		historical_option = yes
		
		set_country_rank_effect = { rank = country_rank:rank_duchy }
		
		add_legitimacy = legitimacy_mild_penalty
		
		hidden_effect = {
			random_owned_location = {
				limit = {
					OR = {
						area = area:upper_rhine_area
						area = area:swabia_area
						area = area:franconia_area
					}
					NOT = { is_connected_to = root.capital }
				}
				save_scope_as = target_order_capital
				
				create_country_from_location = {
					change_government_type = government_type:theocracy
					grant_estate_privilege = estate_privilege:clergy_military_orders
					add_reform = government_reform:military_order_reform
					change_country_name = TEU
					change_country_adjective = TEU_ADJ
					change_country_color = map_TEU
					change_country_flag = TEU
					
					set_capital = scope:target_order_capital
					
					research_advance = advance_type:order_headquarters_advance
					research_advance = advance_type:fortress_church_advance
					
					set_country_rank_effect = { rank = country_rank:rank_duchy }
					
					save_scope_as = target_order_country
				}
			}
			every_owned_location = {
				limit = {
					is_connected_to = scope:target_order_capital
					NOT = { is_connected_to = root.capital }
				}
				change_location_owner = scope:target_order_country
			}
			
			scope:target_order_country = {
				every_province = {
					limit = { any_location_in_province = { NOT = { integration_level = core } } }
					change_province_integration = core
				}
			}
		}
		
	}
	option = {
		name = icd_reformation_push.2.b
	
		add_stability = stability_mild_penalty
	}
}

		}

			save_scope_as = john_calvin
		}
	}
	
	option = { ##calvins triumph##

						fraction = { 0.4 0.55 }

					}
				}
			}
		}
		
		trigger = {
			root = {
				OR = {
					has_or_had_tag = GNV
					num_locations_owned_or_owned_by_subjects_or_below < 20
				}
			}
		}
		hidden_effect = {
			while = {
				count = {
					value = num_characters

				}
				random_character = {
					limit = {
						religion = religion:catholic
					}
					change_character_religion = religion:calvinist
				}
			}
		}
		
		change_religion = religion:calvinist
		change_religion_for_ruler_and_family = { country = ROOT religion = religion:calvinist }
		
		ai_will_select = { value = 400 }
		
		trigger_event_silently = {
			id = icd_reformation_push.10
			months = 2
		}
	}
	
	option = { ##calvins failure##
		name = icd_reformation_push.9.b
		
		scope:john_calvin = {
			move_country = scope:target_exile
		}
	}
	
	option = { ##nothing happens##
		name = icd_reformation_push.9.c
		
		
	}

namespace = icd_reformed

scripted_effect add_to_reformer_list = {
	add_to_global_variable_list = {
		name = reformer_list
		target = scope:target_character
	}
}

icd_reformed.1 = { #andreas karlstadt#
	type = country_event
	category = situation_event
	title = icd_reformed.1.title
	desc = icd_reformed.1.desc
	outcome = neutral

	fire_only_once = yes

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:clergy_estate }

			dip = { 40 70 }
			mil = { 40 70 }

			save_scope_as = target_character
		}
		add_to_reformer_list = yes
		situation:reformation = {
			var:lutheran_character = {
				save_scope_as = target_luther
			}
		}
		
		character:martin_luther = {
			owner ?= { 
				save_scope_as = target_country
			}
		}
		
		hidden_effect = {
			location:leipzig = {
				owner = {
					trigger_event_silently = {
						id = icd_reformed.2
						months = 2
					}
				}
			}
		}
		hidden_effect = {
			character:icd_reformer_andreas_karlstadt = {
				owner = {
					trigger_event_silently = {
						id = icd_reformed.3
						years = 3
					}
				}
			}
		}
	}

	option = {
		name = reformation.1.a
		
		character:icd_reformer_andreas_karlstadt = { 
			move_country = scope:target_country
		}
	}
}

icd_reformed.3 = { # karlstadt changes views#
	type = country_event
	category = situation_event
	title = icd_reformed.3.title
	desc = icd_reformed.3.desc
	outcome = neutral

	fire_only_once = yes

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:clergy_estate }

		character:icd_reformer_andreas_karlstadt = {
			save_scope_as = target_character
		}
	}
	
	trigger = {
		character:icd_reformer_andreas_karlstadt = {
			is_alive = yes
		}
	}

	option = {
		name = icd_reformed.3.a
		
		scope:target_character = {
			change_character_religion = religion:calvinist
			owner ?= {
				capital ?= {
					every_pop = {
						limit = {
							OR = {
								religion = religion:lutheran
								religion = religion:catholic
							}
						}
						split_pop = {
							fraction = 0.1
							religion = religion:calvinist
						}
					}
				}
			}
		}
	}
	
	option = {
		name = icd_reformed.3.b ##has he ever heard of anabaptism#
		
		scope:target_character = {
			change_character_religion = religion:anabaptist
			owner ?= {
				capital ?= {
					every_pop = {
						limit = {
							OR = {
								religion = religion:lutheran
								religion = religion:catholic
							}
						}
						split_pop = {
							fraction = 0.1
							religion = religion:anabaptist
						}
					}
				}
			}
		}
	}

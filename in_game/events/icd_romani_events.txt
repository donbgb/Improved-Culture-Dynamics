icd_romani_events.1 = { ####random migration###
	type = country_event
	title = icd_minority_events.1.title
	desc = icd_minority_events.1.desc
	outcome = neutral
	
	hidden = yes
	
	trigger = {
		
	}
	
	illustration_tags = {
        10 = regular
        10 = exterior
    }
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
		while = {
			count = 3
			random_owned_location = {
				limit = { 
					culture_population = {
						culture = culture:romani_culture
						value > 0.150
					}
				}
				save_scope_as = icd_romani_location_origin
			}
			
			scope:icd_romani_location_origin.market = {
				random_location_in_market = {
					limit = {
						culture_population = {
							culture = culture:romani_culture
							value = 0
						}
						has_owner = yes
					}
					save_scope_as = icd_romani_location_destination
				}
			}
			
			add_migration = {
				 from_location = scope:icd_romani_location_origin
				 to_location = scope:icd_romani_location_destination
				 culture = culture:romani_culture
				 type = pop_type:tribesmen
				 months = 1
				 amount = 0.100
			}
			
			clear_saved_scope = icd_romani_location_origin
			clear_saved_scope = icd_romani_location_destination
		}
		
	}
	
	option = {
		name = icd_minority_events.1.a
		
	}
}

icd_romani_events.99 = { ####when romani arrive###
	type = country_event
	title = icd_romani_events.99.title
	desc = icd_romani_events.99.desc
	outcome = neutral
	
	
	trigger = {
	}
	
	
	illustration_tags = {
        10 = regular
        10 = exterior
    }
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
		
		continent:europe = {
			random_location_in_continent = {
				limit = { 
					culture_population = {
						culture = culture:romani_culture
						value > 0.250
					}
					NOT = {
						owner = {
							has_variable = icd_romani_forbidden_entry
						}
					}
				}
				province_definition = {
					save_scope_as = icd_romani_province_here
				}
			}
		}
	}
	
	option = {
		name = icd_romani_events.99.a ##slave##
		
		grant_estate_privilege = estate_privilege:icd_romani_slaves
	}
	option = {
		name = icd_romani_events.99.b ##anti-romani##
		
		grant_estate_privilege = estate_privilege:icd_anti_romani_laws
	}
	option = {
		name = icd_romani_events.99.c ##craftsmen##
		
		grant_estate_privilege = estate_privilege:icd_romani_craftsmen
	}
	option = {
		name = icd_romani_events.99.d ##expel##
		
		set_variable = icd_romani_forbidden_entry
		
		every_province = {
			limit = {
				culture_population = {
					culture = culture:romani_culture
					value > 0
				}
			}
			add_migration = {
				owner = root
				from = this.province_definition
				to = scope:icd_romani_province_here
				culture = culture:romani_culture
				amount = 2.000
				months = 12
			}
		}
				
	}
	option = {
		name = icd_romani_events.99.e ##cool###
		
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_penalty }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_penalty }
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_penalty }
		
		set_variable = icd_romani_are_chill
	}
}

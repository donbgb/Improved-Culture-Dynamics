icd_romani_events.99 = { ####when romani arrive###
	type = country_event
	title = icd_romani_events.99.title
	desc = icd_romani_events.99.desc
	outcome = neutral
	
	
	trigger = {
	}
	
	
	illustration_tags = {
        10 = regular
        10 = exterior
    }
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
		
		continent:europe = {
			random_location_in_continent = {
				limit = { 
					culture_population = {
						culture = culture:romani_culture
						value > 0.250
					}
					NOT = {
						owner = {
							has_variable = icd_romani_forbidden_entry
						}
					}
				}
				province_definition = {
					save_scope_as = icd_romani_province_here
				}
			}
		}
	}
	
	option = {
		name = icd_romani_events.99.a ##slave##
		
		grant_estate_privilege = estate_privilege:icd_romani_slaves
		
		ai_will_select  = { value = 50 }
	}
	option = {
		name = icd_romani_events.99.b ##anti-romani##
		
		grant_estate_privilege = estate_privilege:icd_anti_romani_laws
		
		ai_will_select  = { value = 100 }
	}
	option = {
		name = icd_romani_events.99.c ##craftsmen##
		
		grant_estate_privilege = estate_privilege:icd_romani_craftsmen
		
		ai_will_select  = { value = 2 }
	}
	option = {
		name = icd_romani_events.99.d ##expel##
		
		set_variable = icd_romani_forbidden_entry
		
		every_province = {
			limit = {
				culture_population = {
					culture = culture:romani_culture
					value > 0
				}
			}
			add_migration = {
				owner = root
				from = this.province_definition
				to = scope:icd_romani_province_here
				culture = culture:romani_culture
				amount = 2.000
				months = 12
			}
		}
		
		ai_will_select  = { value = -20 }
				
	}
	option = {
		name = icd_romani_events.99.e ##cool###
		
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_severe_penalty }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_severe_penalty }
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_severe_penalty }
		
		set_variable = icd_romani_are_chill
		
		ai_will_select  = { value = -700 }
	}
}

icd_romani_events.4 = { ##flavour promising romani###
	type = country_event
	title = icd_romani_events.4.title
	desc = icd_romani_events.4.desc
	outcome = neutral
		
	trigger = {
		culture_population_in_country = {
			culture = culture:romani_culture
			value > 0
		}
		NOT = {
			has_variable = icd_romani_forbidden_entry
		}
	}
	
	fire_only_once = yes
	
	illustration_tags = {
        10 = regular
        10 = interior
    }
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:nobles_estate }
		
		create_character = {
			adm = { 80 95 }
			dip = { 80 95 }
			mil = { 70 80 }
			culture = culture:romani_culture
			religion = root.religion
			age = { 20 30 }
			birth_location = root.capital
			estate = estate_type:burghers_estate
			create_in_limbo = yes
			save_scope_as = icd_romani_cabinet_scope
		}
	}
	
	option = {
		name = icd_romani_events.4.a ##has a place in our court##
		
		add_legitimacy = legitimacy_weak_penalty 
		
		scope:icd_romani_cabinet_scope = {
			move_country = root
		}
	}
	
	option = {
		name = icd_romani_events.4.b ##nothing##

		hidden_effect = {
			kill_character_silently = scope:icd_romani_cabinet_scope
		}
		
	}

}

	type = country_event

		}


icd_global_variable_setup = {
	every_country = {
		limit = {
			OR = {
				NOT = {
					any_pop = {
						culture = prev.culture
						religion != prev.religion
						religion != religion:waldensian
						religion != religion:catharism
						religion != religion:bogomilism
						religion != religion:paulicianism
						pop_size > 0.1
					}
				}
				culture = culture:prussian
				culture = culture:tunisian
				culture = { has_culture_group = culture_group:scandinavian_group }
			}
		}
		set_variable = {
			name = icd_confessional_culture
			value = this.culture
		}
	}
	every_country = {
		limit = { religion.group = religion_group:muslim }
		set_variable = {
			name = icd_waqf_base_var
			value = 10
		}
	}
}

icd_crimean_expel_act = {
	every_province = {
		limit = {
			culture_percentage = {
				culture = culture:crimean
				value > 0.4
			}
			OR = {
				region = region:steppes_region
				region = region:ruthenia_region
			}
		}
		
		add_migration = {
			owner = root
			from = this.province_definition
			to = scope:escape_target1.province_definition
			religion = religion:sunni
			culture = culture:crimean
			amount = {
				value = "this.culture_population(culture:crimean)"
				divide = 4
				divide = 60
			}
			months = 60
		}
		add_migration = {
			owner = root
			from = this.province_definition
			to = scope:escape_target2.province_definition


icd_region_culture_law_german = {
			
			every_province = {
				limit = {
					dominant_culture ?= { has_culture_group = culture_group:german_group }

						dominant_culture ?= culture:frisian
						dominant_culture ?= culture:walloon
						dominant_culture ?= culture:dutch_culture_formed

						dominant_culture ?= culture:czech
					}
					NOT = {
						has_variable = no_reapplying_promotion

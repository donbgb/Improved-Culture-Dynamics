
icd_majority_law_setup = {
	
	every_country = {
		limit = { 
			law:cultural_majority_law = { is_implementable_in = prev } 
			NOT = {
				has_policy = minority_rule
				has_policy = ethnoreligious_integration
				has_policy = dialects_of_german
				has_policy = dialects_of_arabic
				has_policy = ethnic_communities
				has_policy = regional_autonomy
				has_policy = monocultural_opportunity
				has_policy = dialects_of_italian
			}
		}
		if = {
			limit = {
				policy:minority_rule = {
					is_implementable_in = prev 
				}
			}
			add_policy = policy:minority_rule
		}
		else_if = {
			limit = {
				policy:ethnoreligious_integration = {
					is_implementable_in = prev
				}
			}
			add_policy = policy:ethnoreligious_integration
		}
		else_if = {
			limit = {
				policy:dialects_of_german = {
					is_implementable_in = prev
				}
			}
			add_policy = policy:dialects_of_german
		}
		else_if = {
			limit = {
				policy:dialects_of_arabic = {
					is_implementable_in = prev
				}
			}
			add_policy = policy:dialects_of_arabic
		}
		else_if = {
			limit = {
				policy:ethnic_communities = {
					is_implementable_in = prev
				}
			}
			add_policy = policy:ethnic_communities
		}
	}
	

}

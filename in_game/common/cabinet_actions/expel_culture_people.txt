expel_culture_people = { ##expel minority###
	ability = mil
	allow_multiple = no
	
	
	potential = {
		num_locations > 0 
	}
		
	allow = {
		cultural_unity < 1.0
		at_war = no
		in_civil_war = no
	}
	
	months = 2
	
	select_trigger = {
		looking_for_a = culture
		target_flag = target
		#for the AI we have an optimisation to only look through religions that are present. Human can look at all...but will have some disabled
		source_ai_override = actor
		source_flags_ai_override = include_any_present
		name = "expel_culture_people_select_culture"
		none_available_msg_key = "expel_culture_people_no_culture"
		column = {
			data = name
		}
		column = {
			data = culture_percentage_in_country
		}
		visible = {
				scope:actor = {
					culture_percentage_in_country = {
						culture = root
						value > 0
					}
					culture_population_in_country = {
						culture = root
						value > 0.100
					}
				}
				scope:actor = {
					culture_percentage_in_country = {
						culture = root
						value <= 0.1
					}
					NOT = {
						any_owned_location = {
							culture_percentage = {
								culture = root
								value >= 0.9
							}
						}
					}
					NOT = {
						any_province = {
							culture_percentage = {
								culture = root
								value >= 0.8
							}
						}
					}
				}
			NOT = {
				is_primary_or_accepted_in = scope:actor
				is_tolerated_in = scope:actor
			}
		}
	}

	on_activate = {
		custom_tooltip = expel_culture_people_tt
		
		scope:actor = {
			set_variable = {
				name = icd_expel_culture_people
				value = scope:target
			}
		}
	}
	
	is_finished = {
		trigger_if = {
			limit = {
				exists = scope:target
			}
			
			scope:actor = {
				NOT = {
					has_variable = icd_expel_culture_people
				}
			}
		}
	}
	
	on_fully_activated = {
		custom_tooltip = expel_culture_people_activated_tt
		
		scope:actor = {
			trigger_event_silently = {
			 id = icd_minority_events.1000
			 days = 1
			} 
		}
	}
			
	on_deactivate = {
		scope:actor = {
			remove_variable = icd_expel_culture_people
		}
	}


	ai_will_do = {
		value = -100000
	}
}


icd_num_provinces_for_conversion = {
	if = {
		limit = {
			num_locations > 0
			NOT = {
				has_policy = icd_secular_state
				has_policy = icd_yassa_tolerance
				has_policy = icd_sulh_i_kul
				has_policy = icd_custodians
			}
		}
		value = num_provinces
		divide = 10
		min = modifier:government_size
		max = 15
		if = {
			limit = {
				can_see_situation = situation:reformation
				situation:reformation = {
					situation_is_active = yes
				}
			}
			add = 2
		}
		round_to = 1
	}
	
	every_province = {
		if = {
			limit = {
				has_province_modifier = icd_promote_religion_province_modifier
			}
			subtract = 1
		}
	}
}
icd_communal_conversion_rate = {
	value = pop_size
	divide = {
		value = 0
		location = {
			every_pop = {
				limit = {
					owner ?= root
					religion != owner.religion
					culture = root.culture
					pop_type != pop_type:slaves
				}
				add = pop_size
			}
		}
	}
	multiply = 0.15
	
	if = {
		limit = { owner = { modifier:global_pop_conversion_speed_modifier > 0 } }
		add = {
			value = pop_size
			divide = {
				value = 0
				location = {
					every_pop = {
						limit = {
							owner ?= root
							religion != owner.religion
							culture = root.culture
							pop_type != pop_type:slaves
						}
						add = pop_size
					}
				}
			}
			multiply = 0.15
			multiply = owner.modifier:global_pop_conversion_speed_modifier #100% pop conversion is equal to a +0.01% increase#
		}
	}
	if = {
		limit = { location ?= { modifier:local_pop_conversion_speed_modifier > 0 } }
		add = {
			value = pop_size
			divide = {
				value = 0
				location = {
					every_pop = {
						limit = {
							owner ?= root
							religion != owner.religion
							culture = root.culture
							pop_type != pop_type:slaves
						}
						add = pop_size
					}
				}
			}
			multiply = 0.15
			multiply = location.modifier:local_pop_conversion_speed_modifier #100% pop conversion is equal to a +0.01% increase#
		}
	}
	subtract = {
		value = pop_size
		divide = {
			value = 0
			location = {
				every_pop = {
					limit = {
						owner ?= root
						religion != owner.religion
						culture = root.culture
						pop_type != pop_type:slaves
					}
					add = pop_size
				}
			}
		}
		multiply = 0.15
		multiply = {
			value = 1
			subtract = location.local_control
		}
	}
	if = {
		limit = {
			owner ?= {
				OR = {
					has_policy = icd_ethnic_religion
					has_policy = icd_custodians
				}
			}
		}
		multiply = 1.5
	}
	if = {
		limit = {
			pop_type = pop_type:tribesmen
		}
		multiply = 0.5
	}	
}

icd_voluntary_conversion_rate = {
	value = 0.001 ##0.001 equals 0.1%###
	
	if = {
		limit = { pop_satisfaction > 0.25 }
		add = {
			value = pop_satisfaction
			divide = 1000 ##100% pop satisfaction = 0.1%##
		}
	}
	
	if = {
		limit = {
			religion.group != owner.religion.group
		}
		add = {
			value = owner.modifier:tolerance_heathen
			min = 0
			divide = 2000 ##1 tolerance = 0.1%##
		}
	}
	
	if = {
		limit = {
			religion.group = owner.religion.group
		}
		add = {
			value = owner.modifier:tolerance_heretic
			min = 0
			divide = 2000 ##1 tolerance = 0.1%##
		}
	}
	
	if = {
		limit = {
			is_dhimmi = yes
			owner ?= {
				country_has_estate = estate_type:dhimmi_estate
			}
		}
		add = {
			value = "owner.estate(estate_type:dhimmi_estate).num_privileges"
			divide = 4000 ##1 privilege = 0.025%, 4 privileges = 0.1##
		}
	}
	
	if = {
		limit = {
			owner ?= {
				OR = {
					has_policy = icd_secular_state
					has_policy = icd_sulh_i_kul
				}
			}
		}
		add = 0.0015
	}
	
	if = {
		limit = {
			location = {
				area = area:levant_area
			}
			owner ?= {
				is_ai = yes
			}
		}
		divide = 4
	}
	
}

		}
	}
	subtract = {
		value = 1

			value = 1
			subtract = local_control
		}
	}
}

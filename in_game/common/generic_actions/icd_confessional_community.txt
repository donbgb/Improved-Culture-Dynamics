icd_confessional_community = {	
	type = religious

	sound = UI_action_religion_generic
	
	potential = {
		scope:actor = {
			OR = {
				trigger_if = {
					limit = { has_variable = icd_enforced_religion }
					religion = var:icd_enforced_religion
				}
				NOT = {
					has_variable = icd_confessional_culture
				}
				trigger_if = {
					limit = { has_variable = icd_confessional_culture }
					scope:actor.culture != var:icd_confessional_culture
				}
			}
		}
		NOT = { has_game_rule = icd_religious_conversion_vanilla }
		current_year > 1338
	}
	
	allow = {
		scope:actor = {
			OR = {
				AND = {
					custom_tooltip = {
						text = icd_confessional_community_requirements_tt
						NOT = {
							any_pop = {
								culture = scope:actor.culture
								religion != scope:actor.religion
								pop_size > 0.3
							}
						}
					}
					NOT = {
						is_subject_type = fiefdom
						is_subject_type = dominion
						is_subject_type = appanage
					}
				}
				is_colonial_subject = yes
			}
		}
	}
	
	ai_tick = monthly
	ai_tick_frequency = 24
		
	select_trigger = {
		looking_for_a = religion
		source = actor
		target_flag = recipient
		name = "choose_religion"
		column = {
			data = name
		}
	}
	
	show_message = no
	
	effect = {
		if = {
			limit = {
				exists = scope:actor
			}
			scope:actor = {
				custom_tooltip = {
					text = icd_confessional_community_tt
					remove_variable = icd_confessional_culture
					set_variable = {
						name = icd_confessional_culture
						value = scope:actor.culture
					}
					remove_variable = icd_enforced_religion
				}
			}
		}
	}
	
	ai_will_do = {
		value = 200
		
	}
}

western_turkification_culture_movement = {
	culture = turkoman_culture

	potential = {
	}

	allow = {
		custom_tooltip = {
			text = turkification_culture_movement_allow_tt
			always = no
		}
	}

	monthly_spawn_chance = {
		value = 0	
	}
	
	spawn = {
		
	}

	#what's done by spreaders
	environmental_infection = {
		value = 0.2
	}

	required_cultures = { mongolian_culture oirat_culture }
	
	calc_interval_days = 90

	r0 = { 
		value = 0.02
		if = {
			limit = {
				root.owner ?= { 
					culture = { has_culture_group = culture_group:mongolian_group }
					OR = {
						has_policy = assertive_assimilation
						NOT = {
							has_policy = minority_rule
						}
						religion = religion:tengri
						AND = {
							NOT = {
								government_type = government_type:tribe
								government_type = government_type:steppe_horde
							}
							OR = {
								country_rank = country_rank:rank_kingdom
								country_rank = country_rank:rank_empire
							}
						}
					}
				}
			}
			multiply = 0
		}
		if = {
			limit = {
				NOT = { root.region ?= region:crescent_region }
				NOT = { root.region ?= region:caucasus_region }
				NOT = { root.region ?= region:anatolia_region }
				NOT = { root.region ?= region:anatolia_region }
				NOT = { root.area ?= area:iraq_ajam_area }
				NOT = { root.area ?= area:azerbaijan_area }
				NOT = { root.area ?= area:kordestan_area }
				NOT = { root.area ?= area:mazandaran_area }
			}
			multiply = 0
		}
	}


	literacy = negative

	on_spread_to_country = {
	}
	
	specific_pop_type_effect = {
		pop_type = tribesmen
		multiplier = 2.0
	}

	specific_pop_type_effect = {
		religion = tengri
		multiplier = 0.5
	}

	
	on_calc_effect ={
	}

	map_color = {
		if = {
			limit = {
				OR = {
					"culture_percentage(culture:mongolian_culture)" > 0
					"culture_percentage(culture:oirat_culture)" > 0
					"culture_percentage(culture:turkoman_culture)" > 0
				}
				root.owner ?= { 
					culture = { has_culture_group = culture_group:mongolian_group }
					OR = {
						has_policy = assertive_assimilation
						NOT = {
							has_policy = minority_rule
						}
						religion = religion:tengri
						AND = {
							NOT = {
								government_type = government_type:tribe
								government_type = government_type:steppe_horde
							}
							OR = {
								country_rank = country_rank:rank_kingdom
								country_rank = country_rank:rank_empire
							}
						}
					}
				}
			}
			value = define:NMapColors|MAP_COLOR_NULL
		}
		else_if = {
			limit = {
				AND = {
					"culture_percentage(culture:mongolian_culture)" < 0.02
					"culture_percentage(culture:oirat_culture)" < 0.02
				}
				"culture_percentage(culture:turkoman_culture)" > 0.02
			}
			value = define:NMapColors|MAP_COLOR_MAX
		}
		else_if = {
			limit = {
				"culture_percentage(culture:turkoman_culture)" > "culture_percentage(culture:mongolian_culture)"
				"culture_percentage(culture:turkoman_culture)" > "culture_percentage(culture:oirat_culture)"
			}
			value = define:NMapColors|MAP_COLOR_HIGH
		}
		else_if = {
			limit = {
				"culture_percentage(culture:turkoman_culture)" > 0
			}
			value = define:NMapColors|MAP_COLOR_MID
		}
		else = {
			value = define:NMapColors|MAP_COLOR_LOW
		}
	}
	
	location_spread_threshold = {
		value = 0.05 # % of the people need to be infected to spread to neighbors
	}
	
	

}

dutch_culture_movement = {
	culture = dutch_culture_formed

	potential = {
	}

	allow = {
		custom_tooltip = {
			text = dutch_culture_movement_allow_tt
			always = no
		}
	}

	monthly_spawn_chance = {
		value = 0
		
		
	}
	
	spawn = {
		ordered_ownable_location = {
			limit = { OR = { owner = { has_or_had_tag = NED } owner = { is_subject_or_below_of = c:NED } } }
			max = 1

			order_by = {
				value = population				
			}
			spawn_movement = {
				movement_definition = scope:movement_definition
				supporters = {
					value = population
					multiply = 0.15
				}
			}
		}
	}

	#what's done by spreaders
	environmental_infection = {
		value = 0.05
	}

	required_cultures = { frisian picard walloon westphalian low_franconian }
	

	r0 = { 
		value = { 0.85 0.95 }
		
		if = {
			limit = {
				NOT = { root.owner ?= { has_or_had_tag = NED } }
				NOT = { root.owner ?= { is_subject_or_below_of = c:NED } }
			}
			multiply = 0.0
		}
		
	}

	calc_interval_days = 30

	development = positive
	literacy = positive
	pop_satisfaction = positive

	on_spread_to_country = {
	}

	specific_pop_type_effect = {
		pop_type = tribesmen
		multiplier = 0.01
	}

	specific_pop_type_effect = {
		pop_type = tribesmen
		multiplier = 0.01
	}
	
	specific_pop_type_effect = {
		culture = picard
		multiplier = 0.1
	}
	
	specific_pop_type_effect = {
		culture = walloon
		multiplier = 0.25
	}

	location_stagnation_chance = {
		if = {
			limit = {
				dominant_culture ?= culture:dutch_culture_formed
			}
			value = 1
		}
		if = {
			limit = {
				has_location_modifier = dutch_culture_movement_momentum_waning_locally
			}
			value = 1
		}
	}
	
	on_calc_effect ={
		every_location_with_movement = {
			limit = {
				NOT = {
					dominant_culture ?= culture:dutch_culture_formed
				}
			}
			add_location_modifier = {
				modifier = dutch_culture_movement_momentum_waning_locally
				size = 0.34
				mode = add_and_extend
			}
		}
	}

	map_color = {
		if = {
			limit = {
				location_modifier_strength = {
					modifier = dutch_culture_movement_momentum_waning_locally
					value > 0.9
				}
			}
			value = define:NMapColors|MAP_COLOR_NULL 
		}
		else_if = {
			limit = {
				"culture_percentage(culture:dutch_culture_formed)" > 0.05
			}
			value = map_NED
		}
		else = {
			value = define:NMapColors|MAP_COLOR_LOW
		}
	}
	
	location_spread_threshold = {
		value = 0.5 # % of the people need to be infected to spread to neighbors
	}
	
	

}

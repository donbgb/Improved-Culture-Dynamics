
on_game_start = {
	
	on_actions = {
		delay = { days =  1 }
		icd_intro_event_action
	}
}

icd_intro_event_action = {
	trigger = {
		always = yes
	}
	
	effect = {
		every_country = {
			limit = { 
				is_ai = no
				has_game_rule = icd_intro_event_on
			}
			trigger_event_silently = icd_intro_event.1
		}
	}
}

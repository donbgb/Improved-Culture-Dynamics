
on_game_start = {
	
	on_actions = {
		icd_culture_on_action
		delay = { days =  1 }
		icd_culture_on_action_majority
	}
}

on_character_created = {
	
	on_actions = {
		icd_character_culture_on_action
	}

}

icd_culture_on_action = {
	trigger = {
		always = yes
	}
	
	effect = {
		icd_cultural_setup = yes
		icd_religion_setup = yes
		icd_remove_accepted_tolerated_cultures = yes
		icd_sects_setup = yes
		icd_movements_setup = yes
		icd_politics_setup = yes
		icd_romani_setup = yes
		icd_religious_view_setup = yes
		icd_create_nizari_imamate = yes
	}

}
icd_culture_on_action_majority = {
	trigger = {
		always = yes
	}
	
	effect = {
		icd_majority_law_setup = yes
		icd_global_variable_setup = yes
	}

}
icd_character_culture_on_action = {
	trigger = {
		always = yes
	}
	
	effect = {
		icd_character_culture_change = yes
	}

}


on_new_age = {
		
	on_actions = {
		icd_romani_on_new_age
		icd_destroy_movements
	}
		
}

icd_romani_on_new_age = {
	
	events = {
		icd_romani_events.2
		icd_romani_events.3
	}

}
icd_destroy_movements = {
	trigger = {
		current_age = age_4_reformation
	}
	
	effect = {
		culture:turkoman_culture = {
			every_movement_in_culture = {
				destroy_movement = yes
			}
		}
		culture:turkmen_culture = {
			every_movement_in_culture = {
				destroy_movement = yes
			}
		}
		religion:shia = {
			every_movement_in_religion = {
				destroy_movement = yes
			}
		}
		religion:shia = {
			every_religious_order_in_religion = {
				limit = {
					order_type ?= religious_order_type:icd_militant_sufi_order
				}
				every_location_in_religious_order = {
					remove_religious_order = prev
				}
			}
		}
		every_country = {
			limit = { religion.group = religion_group:muslim }
			trigger_event_silently = icd_muslim_events.26
		}
	}
}

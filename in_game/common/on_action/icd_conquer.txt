
on_location_changed_owner = {

	on_actions = {
		icd_stop_religion
		icd_caliph_change
	}
}

icd_stop_religion = {
	
	effect = {
		icd_conversion_stop = yes
	}
}

icd_caliph_change = {
	
	effect = {
		if = {
			limit = {
				this = location:cairo
				scope:loser = { 
					has_variable = icd_abbasid_caliph
				}
			}
			scope:loser = {
				remove_variable = icd_abbasid_caliph
				international_organization_remove_special_status = {
					type = special_status:icd_caliphate_claimant
					country = scope:actor
				}
			}
			international_organization_chooses_new_leader = international_organization:icd_sunni_caliphate
		}
	}
}

on_character_death = {

	on_actions = {
		icd_caliph_check_abbasid
	}
}
icd_caliph_check_abbasid = {
	trigger = {
		scope:target.owner = {
			has_variable = icd_abbasid_caliph
			scope:target = var:icd_abbasid_caliph
		}
	}
	
	effect = {
		dynasty:icd_abbasid_dynasty = {
			ordered_character_in_dynasty = {
				limit = {
					is_alive = yes
					is_female = no
				}
				order_by = age_in_years
				scope:target.owner = {
					remove_variable = icd_abbasid_caliph
					set_variable = {
						name = icd_abbasid_caliph
						value = prev
					}
				}
			}
		}
	}
}

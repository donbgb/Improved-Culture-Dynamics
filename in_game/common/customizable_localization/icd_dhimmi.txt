
icd_dhimmi_communities_each_israelite = {
	type = country
	
	text = {
		trigger = {
			NOT = { has_variable = icd_israelite_dhimmi_disabled }
			religion_group_population_in_country = {
				religion_group = religion_group:israelite_group
				value > 0.010
			}
		}
		localization_key = icd_dhimmi_communities_each_israelite_custom
	}
}

icd_dhimmi_communities_each_sabian = {
	type = country
	
	text = {
		trigger = {
			NOT = { has_variable = icd_sabian_dhimmi_disabled}

				religion_group_population_in_country = {
					religion_group = religion_group:mandean_group
					value > 0.010
				}

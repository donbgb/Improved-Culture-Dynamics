icd_dhimmi_communities_protection = {
	icon = dhimmi_abrahamic_communities
	estate = dhimmi_estate
	potential = {
		religion ?= {
			group ?= religion_group:muslim
		}
	}
	
	can_revoke = {
		NOT = { 
			has_policy = ethnoreligious_integration
		}
		hidden_trigger = {
			OR = {
				is_human = yes
				religious_unity < 0.3
			}
		}
		NOT = {
			custom_tooltip = { 
				text = icd_dhimmi_communities_revoke_tt
				OR = {
					NOT = { has_variable = icd_israelite_dhimmi_disabled }
					NOT = { has_variable = icd_christian_dhimmi_disabled }
					NOT = { has_variable = icd_sabian_dhimmi_disabled}
					NOT = { has_variable = icd_iranic_dhimmi_disabled }
					NOT = { has_variable = icd_dharmic_dhimmi_disabled }
					NOT = { has_variable = icd_buddhist_dhimmi_disabled }
				}
			}
		}
	}
	allow = {
		custom_tooltip = { 
			text = icd_dhimmi_communities_allow_tt
			OR = {
				NOT = { has_variable = icd_israelite_dhimmi_disabled }
				NOT = { has_variable = icd_christian_dhimmi_disabled }
				NOT = { has_variable = icd_sabian_dhimmi_disabled}
				NOT = { has_variable = icd_iranic_dhimmi_disabled }
				NOT = { has_variable = icd_dharmic_dhimmi_disabled }
				NOT = { has_variable = icd_buddhist_dhimmi_disabled }
			}
		}
	}
	
	
	location_modifier = {
		potential_trigger = {
			custom_tooltip = {
				text = icd_dhimmi_communities_trigger_tt
				OR = {
					dominant_religion ?= { 
							group = religion_group:muslim
						}
					AND = {
						dominant_religion ?= { 
							group = religion_group:israelite_group
						}
						owner ?= { NOT = { has_variable = icd_israelite_dhimmi_disabled } }
					}
					AND = {
						dominant_religion ?= { 
							group = religion_group:christian
						}
						owner ?= { NOT = { has_variable = icd_christian_dhimmi_disabled } }
					}
					AND = {
						dominant_religion ?= { 
							group = religion_group:mandean_group
						}
						owner ?= { NOT = { has_variable = icd_sabian_dhimmi_disabled} }
					}
					AND = {
						dominant_religion ?= { 
							group = religion_group:manichaean_group
						}
						owner ?= { NOT = { has_variable = icd_sabian_dhimmi_disabled} }
					}
					AND = {
						dominant_religion ?= { 
							group = religion_group:zoroastrian_group
						}
						owner ?= { NOT = { has_variable = icd_iranic_dhimmi_disabled } }
					}
					AND = {
						dominant_religion ?= { 
							group = religion_group:dharmic
						}
						owner ?= { NOT = { has_variable = icd_dharmic_dhimmi_disabled } }
					}
					AND = {
						dominant_religion ?= { 
							group = religion_group:buddhist
						}
						owner ?= { NOT = { has_variable = icd_buddhist_dhimmi_disabled } }
					}
				}
				OR = {
					AND = {
						religion_group_percentage = {
							religion_group = religion_group:israelite_group
							value > 0.01
						}
						owner ?= { NOT = { has_variable = icd_israelite_dhimmi_disabled } }
					}
					AND = {
						religion_group_percentage = {
							religion_group = religion_group:christian
							value > 0.01
						}
						owner ?= { NOT = { has_variable = icd_christian_dhimmi_disabled } }
					}
					
					AND = {
						religion_group_percentage = {
							religion_group = religion_group:mandean_group
							value > 0.01
						}
						owner ?= { NOT = { has_variable = icd_sabian_dhimmi_disabled} }
					}
					AND = {
						religion_group_percentage = {
							religion_group = religion_group:manichaean_group
							value > 0.01
						}
						owner ?= { NOT = { has_variable = icd_sabian_dhimmi_disabled} }
					}
					AND = {
						religion_group_percentage = {
							religion_group = religion_group:zoroastrian_group
							value > 0.01
						}
						owner ?= { NOT = { has_variable = icd_iranic_dhimmi_disabled } }
					}
					AND = {
						religion_group_percentage = {
							religion_group = religion_group:dharmic
							value > 0.01
						}
						owner ?= { NOT = { has_variable = icd_dharmic_dhimmi_disabled } }
					}
					AND = {
						religion_group_percentage = {
							religion_group = religion_group:buddhist
							value > 0.01
						}
						owner ?= { NOT = { has_variable = icd_buddhist_dhimmi_disabled } }
					}
				}
				religion_group_percentage = {
					religion_group = religion_group:folk_asian_group
					value = 0
				}
				religion_group_percentage = {
					religion_group = religion_group:folk_european_group
					value = 0
				}
				religion_group_percentage = {
					religion_group = religion_group:folk_se_asian_group
					value = 0
				}
			}
		}
		local_heathen_pop_conversion_speed_modifier = -0.5
	}
	
	location_modifier = {
		potential_trigger = {
			integration_level = core
		}
		building_enslavement_power = -0.100
	}
	
	country_modifier = {
		dhimmi_estate_target_satisfaction = small_privilege_target_satisfaction
		global_dhimmi_estate_power = 1
	}
	country_modifier = {
		tolerance_heathen = 0.25
	}
	
}

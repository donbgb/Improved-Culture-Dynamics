icd_vlach_law = {
	estate = peasants_estate
	icon = tribes_pasture_access

	potential = {
		NOT = {
			culture = { has_culture_group = culture_group:romanian_group }
		}
		any_pop = { 
			OR = {
				culture = { has_culture_group = culture_group:romanian_group }
			}
		}
	}
	
	can_revoke = {
		NOT = {
			any_accepted_culture = {
				custom_tooltip = { 
					text = icd_romanian_language_tt
					language = language:romanian_language
				}
			}
			any_tolerated_culture = {
				custom_tooltip = { 
					text = icd_romanian_language_tt
					language = language:romanian_language
				}
			}
		}
	}

	country_modifier = {
		global_livestock_output_modifier = 0.05
		global_wool_output_modifier = 0.05
		global_pop_food_consumption = -0.01
		peasants_estate_target_satisfaction = small_privilege_target_satisfaction
		global_peasants_estate_power = 0.1
	}
	
	location_modifier = {
		potential_trigger = {
			OR = {
				dominant_culture.language ?= language:romanian_language
				culture_group_percentage = {
					culture_group = culture_group:romanian_group
					value > 0.13
				}
			}
		}
		local_pop_assimilation_speed_modifier = -0.9
		
	}
	years = 2
}


icd_romani_slaves = {
	estate = nobles_estate
	icon = icd_romani_slaves

	potential = {
		NOT = {
			culture = { has_culture_group = culture_group:romani_group }
		}
		any_pop = { 
			AND = {
				culture = culture:romani_culture
				pop_type = pop_type:tribesmen
			}
		}
	}
	
	allow = {
		NOT = {
			has_estate_privilege = estate_privilege:icd_anti_romani_laws
			has_estate_privilege = estate_privilege:icd_romani_craftsmen
		}
	}
	
	on_deactivate = {
		if = {
			limit = {
				any_pop = { 
					culture = culture:romani_culture
				}
			}
			set_variable = icd_romani_are_chill
		}
	}
	
	can_revoke = {
		hidden_trigger = {
			is_ai = no
		}
	}
	
	on_activate = {
		if = {
			limit = {
				has_variable = icd_romani_are_chill
			}
			remove_variable = icd_romani_are_chill
		}
	}

	country_modifier = {
		nobles_estate_target_satisfaction = small_privilege_target_satisfaction
		global_nobles_estate_power = 0.01
	}
	
	location_modifier = {
		potential_trigger = {
			custom_tooltip = {
				text = icd_romani_privilege_tt
				culture_group_population = {
					culture_group = culture_group:romani_group
					value > 0
				}
				any_pop = {
					culture = { has_culture_group = culture_group:romani_group }
					pop_type = pop_type:tribesmen
				}
			}
		}
		local_tribal_promotion = -6.0
		local_raw_material_output = 0.025
	}
	years = 2
}

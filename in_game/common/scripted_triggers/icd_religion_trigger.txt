is_protected_pop = {
	OR = {
		AND = {
			religion.group = religion_group:israelite_group
			owner ?= {
				OR = {
					has_policy = icd_aljama_communities
					has_policy = icd_jewish_charter
					has_policy = icd_jewish_ghetto
					has_policy = council_of_three_lands_policy
					has_policy = icd_subjugated_communities
				}
			}
			location.dominant_religion.group = owner.religion.group
		}
		AND = {
			religion.group = religion_group:muslim
			owner ?= {
				OR = {
					has_policy = icd_aljama_communities
					has_policy = icd_subjugated_communities
				}
			}
			location.dominant_religion.group = owner.religion.group
		}
		AND = {
			is_dhimmi = yes
			owner ?= {
				religion.group = religion_group:muslim
			}
		}
		AND = {
			owner ?= {
				modifier:tolerance_heretic >= -1.5
			}
			religion != owner.religion
			religion.group = owner.religion.group
			owner.religion = { religious_view = { target = prev.religion value = kindred } }	
		}
		AND = {
			owner ?= {
				modifier:tolerance_heretic >= -0.5
			}
			religion != owner.religion
			religion.group = owner.religion.group
			owner.religion = { religious_view = { target = prev.religion value = positive } }	
		}
		AND = {
			owner ?= {
				modifier:tolerance_heretic >= 0
			}
			religion != owner.religion
			religion.group = owner.religion.group
			owner.religion = { religious_view = { target = prev.religion value = neutral } }	
		}
		AND = {
			owner ?= {
				modifier:tolerance_heretic >= 0.5
			}
			religion != owner.religion
			religion.group = owner.religion.group
			owner.religion = { religious_view = { target = prev.religion value >= negative } }	
		}
		AND = {
			owner ?= {
				modifier:tolerance_heretic >= 1.5
			}
			religion != owner.religion
			religion.group = owner.religion.group
			owner.religion = { religious_view = { target = prev.religion value = enemy } }	
		}
		
		AND = {
			owner ?= {
				modifier:tolerance_heathen >= -1.5
			}
			religion.group != owner.religion.group
			owner.religion = { religious_view = { target = prev.religion value = kindred } }	
		}
		AND = {
			owner ?= {
				modifier:tolerance_heathen >= -0.5
			}
			religion.group != owner.religion.group
			owner.religion = { religious_view = { target = prev.religion value = positive } }	
		}
		AND = {
			owner ?= {
				modifier:tolerance_heathen >= 0
			}
			religion.group != owner.religion.group
			owner.religion = { religious_view = { target = prev.religion value = neutral } }	
		}
		AND = {
			owner ?= {
				modifier:tolerance_heathen >= 0.5
			}
			religion.group != owner.religion.group
			owner.religion = { religious_view = { target = prev.religion value >= negative } }	
		}
		AND = {
			owner ?= {
				modifier:tolerance_heathen >= 1.5
			}
			religion.group != owner.religion.group
			owner.religion = { religious_view = { target = prev.religion value = enemy } }	
		}
		
		AND = {
			owner ?= {
				modifier:global_burghers_conversion_blocked = yes
			}
			pop_type = pop_type:burghers
		}
		AND = {
			owner ?= {
				modifier:global_clergy_conversion_blocked = yes
			}
			pop_type = pop_type:clergy
		}
		AND = {
			owner ?= {
				modifier:global_laborers_conversion_blocked = yes
			}
			pop_type = pop_type:laborers
		}
		AND = {
			owner ?= {
				modifier:global_nobles_conversion_blocked = yes
			}
			pop_type = pop_type:nobles
		}
		AND = {
			owner ?= {
				modifier:global_peasants_conversion_blocked = yes
			}
			pop_type = pop_type:peasants
		}
		AND = {
			owner ?= {
				modifier:global_soldiers_conversion_blocked = yes
			}
			pop_type = pop_type:soldiers
		}
		AND = {
			owner ?= {
				modifier:global_tribesmen_conversion_blocked = yes
			}
			pop_type = pop_type:tribesmen
		}
	}
}

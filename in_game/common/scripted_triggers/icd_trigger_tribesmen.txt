icd_is_steppe_tribesmen = {
	owner ?= {
		OR = {
			government_type = government_type:steppe_horde
			government_type = government_type:tribe
		}
		num_locations > 0
	}
	OR = {	
		culture = owner.culture
		culture = { has_culture_group = culture_group:mongolian_group }
		culture = { has_culture_group = culture_group:tatar_group }
		culture = { has_culture_group = culture_group:steppe_group }
	}
}

## Not hardcoded
# Modifiers in this file will automatically be applied to all countries
#	- Tooltips will also be generated
#	- The AI will also understand them


icd_conversion_less = {
	
	type = country
	
	potential_trigger = {
		NOT = { has_game_rule = icd_religious_conversion_vanilla }
	}
	
	hide_effects = yes
	alert = no
	requires_real = yes
	
	global_pop_conversion_speed = -2.0
}



icd_tolerance_heretic = {
	type = country
	
	potential_trigger = {
		modifier:tolerance_heretic >= -1.0
		has_game_rule = icd_religious_conversion_vanilla
	}
	
	scales_with = {
		value = icd_religion_tolerance_heretic
	} 
	
	hide_effects = no
	alert = no
	requires_real = no # Non-"real" countries are Rebels, Pirates and Mercenaries. Default is yes.

	global_heretic_pop_conversion_speed_modifier = -0.1
}

						NOT = { is_member_of_international_organization = international_organization:icd_nizari_imamate_io }

				}
				every_country = {
					limit = {
						is_rebel_country = no
						religion = religion:nizari_religion
						NOT = { is_member_of_international_organization = international_organization:icd_nizari_imamate_io }
					}
					root = { add_country_to_international_organization = prev }
				}
			}
		}

			limit = {
				current_year > 1340
				international_organization_has_leader = no
				NOT = { international_organization_has_policy = policy:icd_sadr }

			add_policy_to_international_organization = policy:icd_sadr

				trigger_event_silently = icd_muslim_events.28
			}
		}
		else_if = {
			limit = {
				current_year > 1750

			add_policy_to_international_organization = policy:icd_aga_khan
			every_international_organization_member = {
				trigger_event_silently = icd_muslim_events.29
			}
		}
		else_if = {
			limit = {
				any_international_organization_member = {
					country_rank_level > 2
				}
				NOT = { international_organization_has_policy = policy:icd_ascension_stance }
			}
			add_policy_to_international_organization = policy:icd_ascension_stance
			every_international_organization_member = {
				trigger_event_silently = icd_muslim_events.30
			}
		}
		else_if = {
			limit = {
				any_international_organization_member = {
					current_year > 1400
					any_province = {
						dominant_religion = religion:nizari_religion
					}
				}
				NOT = { international_organization_has_policy = policy:icd_revival_stance }
			}
			add_policy_to_international_organization = policy:icd_revival_stance
			every_international_organization_member = {
				trigger_event_silently = icd_muslim_events.31
			}
		}
		else_if = {
			limit = { NOT = { international_organization_has_policy = policy:icd_taqiyya_stance } }

			every_international_organization_member = {

		}
